   153→// 💰 نمایش مبلغ — همیشه «$X.XX USD»؛ مقدار ناموجود = —
   154→function fmtUsd(v: string | number | null | undefined): string {
   155→  if (v === null || v === undefined || v === '') return '—'
   156→  const n = Number(v)
   157→  return Number.isFinite(n) ? `$${n.toFixed(2)} USD` : '—'
   158→}

   160→function fmtDateTime(iso: string | null | undefined): string {
   161→  if (!iso) return '—'
   162→  const d = new Date(iso)
   163→  if (Number.isNaN(d.getTime())) return '—'
   164→  return `${d.toLocaleDateString()} ${d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}`
   165→}
   166→
   167→function fmtDate(iso: string | null | undefined): string {
   168→  if (!iso) return '—'
   169→  const d = new Date(iso)
   170→  return Number.isNaN(d.getTime()) ? '—' : d.toLocaleDateString()
   171→}

   196→export function PaymentsAdminTab({
   197→  token,
   198→  onToast,
   199→  initialFilter,
   200→  onCounts,
   201→  onChanged,
   202→}: {
   203→  token: string
   204→  onToast?: (msg: string) => void
   205→  initialFilter?: string
   206→  onCounts?: (c: PayCounts) => void
   207→  onChanged?: () => void
   208→}) {
   209→  // ----- فهرست -----
   210→  const [filter, setFilter] = useState(initialFilter && initialFilter.trim() !== '' ? initialFilter : 'receipt_submitted')
   211→  const [query, setQuery] = useState('')
   212→  const [debouncedQ, setDebouncedQ] = useState('')
   213→  const [page, setPage] = useState(1)
   214→  const [data, setData] = useState<PayListData | null>(null)
   215→  const [listLoading, setListLoading] = useState(true)
   216→  const [listError, setListError] = useState(false)
   217→
   218→  // کال‌بک‌های والد در ref نگه داشته می‌شوند تا شناسه‌شان حلقهٔ واکشی نسازد
   219→  const cbRef = useRef({ onCounts, onChanged })
   220→  useEffect(() => {
   221→    cbRef.current = { onCounts, onChanged }
   222→  })
   223→
   224→  // debounce جست‌وجو (۳۰۰ms)
   225→  useEffect(() => {
   226→    const t = setTimeout(() => setDebouncedQ(query.trim()), 300)
   227→    return () => clearTimeout(t)
   228→  }, [query])
   229→
   230→  // الگوی رسمی ریکت: بازگشت به صفحهٔ ۱ با تغییر فیلتر/جست‌وجو (بدون setState داخل effect)
   231→  const [prevListKey, setPrevListKey] = useState('')
   232→  const listKey = `${filter}|${debouncedQ}`
   233→  if (prevListKey !== listKey) {
   234→    setPrevListKey(listKey)
   235→    setPage(1)
   236→  }
   237→
   238→  // سینک فیلتر با پراپ (باز شدن تب از داشبورد با فیلتر مقصد)
   239→  useEffect(() => {
   240→    if (typeof initialFilter === 'string' && initialFilter !== '') {
   241→      setFilter(initialFilter)
   242→    }
   243→  }, [initialFilter])
   244→
   245→  const loadList = useCallback(async () => {
   246→    setListLoading(true)
   247→    setListError(false)
   248→    try {
   249→      const params = new URLSearchParams({ status: filter, page: String(page), pageSize: String(PAGE_SIZE) })
   250→      if (debouncedQ) params.set('q', debouncedQ)
   251→      const res = await fetch(`/api/admin/payments?${params.toString()}`, {
   252→        headers: { 'x-admin-key': token },
   253→        cache: 'no-store',
   254→      })
   255→      if (!res.ok) {
   256→        setListError(true)
   257→        setData(null)
   258→        return
   259→      }
   260→      const d = (await res.json()) as PayListData
   261→      setData(d)
   262→      if (d.counts) cbRef.current.onCounts?.(d.counts)
   263→    } catch {
   264→      setListError(true)
   265→      setData(null)
   266→    } finally {
   267→      setListLoading(false)
   268→    }
   269→  }, [token, filter, page, debouncedQ])
   270→
   271→  useEffect(() => {
   272→    void loadList()
   273→  }, [loadList])
   274→
   275→  // ----- دیالوگ جزئیات -----
   276→  const [detailOpen, setDetailOpen] = useState(false)
   277→  const [detailRef, setDetailRef] = useState<string | null>(null)
   278→  const [detail, setDetail] = useState<PayDetailData | null>(null)
   279→  const [detailLoading, setDetailLoading] = useState(false)
   280→  const [detailError, setDetailError] = useState(false)
   281→  const [actionBusy, setActionBusy] = useState(false)
   282→  const [approveStage, setApproveStage] = useState(false)
   283→  const [rejectStage, setRejectStage] = useState(false)
   284→  const [rejectReason, setRejectReason] = useState('')
   285→
   286→  const loadDetail = useCallback(
   287→    async (ref: string) => {
   288→      setDetailLoading(true)
   289→      setDetailError(false)
   290→      try {
   291→        const res = await fetch(`/api/admin/payments/${encodeURIComponent(ref)}`, {
   292→          headers: { 'x-admin-key': token },
   293→          cache: 'no-store',
   294→        })
   295→        if (!res.ok) {
   296→          setDetailError(true)
   297→          setDetail(null)
   298→          return
   299→        }
   300→        setDetail((await res.json()) as PayDetailData)
   301→      } catch {
   302→        setDetailError(true)
   303→        setDetail(null)
   304→      } finally {
   305→        setDetailLoading(false)
   306→      }
   307→    },
   308→    [token]
   309→  )
   310→
   311→  const openDetail = (ref: string) => {
   312→    setDetailRef(ref)
   313→    setDetail(null)
   314→    setDetailError(false)
   315→    setApproveStage(false)
   316→    setRejectStage(false)
   317→    setRejectReason('')
   318→    setDetailOpen(true)
   319→    void loadDetail(ref)
   320→  }
   321→
   322→  const closeDetail = () => {
   323→    setDetailOpen(false)
   324→    setDetailRef(null)
   325→    setDetail(null)
   326→    setApproveStage(false)
   327→    setRejectStage(false)
   328→    setRejectReason('')
   329→    setLightbox(false)
   330→  }
   331→
   332→  // 🧾 واکشی رسید با هدر ادمین → Object URL (با بستن/تعویض رسید revoke می‌شود)
   333→  const receiptSrc = detailOpen ? (detail?.order.receiptUrl ?? null) : null
   334→  const [receiptObjUrl, setReceiptObjUrl] = useState<string | null>(null)
   335→  const [receiptState, setReceiptState] = useState<'idle' | 'loading' | 'ready' | 'error'>('idle')
   336→  const [lightbox, setLightbox] = useState(false)
   337→  useEffect(() => {
   338→    let cancelled = false
   339→    let created: string | null = null
   340→    setReceiptObjUrl(null)
   341→    setLightbox(false)
   342→    if (!receiptSrc) {
   343→      setReceiptState('idle')
   344→      return
   345→    }
   346→    setReceiptState('loading')
   347→    fetch(receiptSrc, { headers: { 'x-admin-key': token }, cache: 'no-store' })
   348→      .then((res) => {
   349→        if (!res.ok) throw new Error('receipt fetch failed')
   350→        return res.blob()
   351→      })
   352→      .then((blob) => {
   353→        const url = URL.createObjectURL(blob)
   354→        if (cancelled) {
   355→          URL.revokeObjectURL(url)
   356→          return
   357→        }
   358→        created = url
   359→        setReceiptObjUrl(url)
   360→        setReceiptState('ready')
   361→      })
   362→      .catch(() => {
   363→        if (!cancelled) setReceiptState('error')
   364→      })
   365→    return () => {
   366→      cancelled = true
   367→      if (created) URL.revokeObjectURL(created)
   368→    }
   369→  }, [receiptSrc, token])
   370→
   371→  const refreshAll = useCallback(
   372→    async (ref: string) => {
   373→      await Promise.all([loadDetail(ref), loadList()])
   374→      cbRef.current.onChanged?.()
   375→    },
   376→    [loadDetail, loadList]
   377→  )
   378→
   379→  // ----- اقدامات -----
   380→  const doApprove = async () => {
   381→    if (!detailRef || actionBusy) return
   382→    setActionBusy(true)
   383→    try {
   384→      const res = await fetch(`/api/admin/payments/${encodeURIComponent(detailRef)}/approve`, {
   385→        method: 'POST',
   386→        headers: { 'Content-Type': 'application/json', 'x-admin-key': token },
   387→        body: JSON.stringify({}),
   388→      })
   389→      const d = (await res.json().catch(() => ({}))) as { error?: string }
   390→      if (!res.ok) {
   391→        onToast?.(d.error || pa.approveFailed)
   392→        return
   393→      }
   394→      onToast?.(pa.approvedToast)
   395→      setApproveStage(false)
   396→      await refreshAll(detailRef)
   397→    } catch {
   398→      onToast?.(pa.approveFailed)
   399→    } finally {
   400→      setActionBusy(false)
   401→    }
   402→  }
   403→
   404→  const doReject = async () => {
   405→    const reason = rejectReason.trim()
   406→    if (!detailRef || actionBusy || reason.length < 3) return
   407→    setActionBusy(true)
   408→    try {
   409→      const res = await fetch(`/api/admin/payments/${encodeURIComponent(detailRef)}/reject`, {
   410→        method: 'POST',
   411→        headers: { 'Content-Type': 'application/json', 'x-admin-key': token },
   412→        body: JSON.stringify({ reason }),
   413→      })
   414→      const d = (await res.json().catch(() => ({}))) as { error?: string }
   415→      if (!res.ok) {
   416→        onToast?.(d.error || pa.rejectFailed)
   417→        return
   418→      }
   419→      onToast?.(pa.rejectedToast)
   420→      setRejectStage(false)
   421→      setRejectReason('')
   422→      await refreshAll(detailRef)
   423→    } catch {
   424→      onToast?.(pa.rejectFailed)
   425→    } finally {
   426→      setActionBusy(false)
   427→    }
   428→  }
   429→
   430→  const doCancel = async () => {
   431→    if (!detailRef || actionBusy) return
   432→    if (!window.confirm(a.cancelConfirm)) return
   433→    setActionBusy(true)
   434→    try {
   435→      const res = await fetch(`/api/payments/orders/${encodeURIComponent(detailRef)}`, {
   436→        method: 'PATCH',
   437→        headers: { 'Content-Type': 'application/json', 'x-admin-key': token },
   438→        body: JSON.stringify({ action: 'cancel' }),
   439→      })
   440→      const d = (await res.json().catch(() => ({}))) as { error?: string }
   441→      if (!res.ok) {
   442→        onToast?.(d.error || pa.loadFailed)
   443→        return
   444→      }
   445→      onToast?.(a.orderCancelled)
   446→      await refreshAll(detailRef)
   447→    } catch {
   448→      onToast?.(pa.loadFailed)
   449→    } finally {
   450→      setActionBusy(false)
   451→    }
   452→  }
   453→
   454→  const counts = data?.counts ?? null
   455→  const total = data?.total ?? 0
   456→  const totalPages = Math.max(1, Math.ceil(total / PAGE_SIZE))
   457→  const orders = data?.orders ?? []
   458→
   459→  // کارت‌های خلاصه — کلیک = فیلتر همان وضعیت
   460→  const summaryCards: { key: string; label: string; value: number; icon: React.ReactNode; tone: string }[] = [

   473→      tone: 'bg-green-100 text-green-700',
   474→    },
   475→    {
   476→      key: 'rejected',
   477→      label: pa.summaryRejected,
   478→      value: counts?.rejected ?? 0,
   479→      icon: <XCircle className="w-5 h-5" aria-hidden="true" />,
   480→      tone: 'bg-peach-light/60 text-brown',
   481→    },
   482→    {
   483→      key: 'unpaid',
   484→      label: pa.summaryUnpaid,
   485→      value: counts?.unpaid ?? 0,
   486→      icon: <Hourglass className="w-5 h-5" aria-hidden="true" />,
   487→      tone: 'bg-butter/40 text-brown',
   488→    },
   489→  ]
   490→
   491→  const isReviewable = detail?.order.status === 'RECEIPT_SUBMITTED'
   492→  const isImageReceipt = (detail?.order.receiptMime ?? '').startsWith('image/')
   493→
   494→  return (
   495→    <div>
   496→      {/* ---------- کارت‌های خلاصه ---------- */}
   497→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-5">
   498→        {summaryCards.map((c) => (
   499→          <button
   500→            key={c.key}
   501→            type="button"
   502→            onClick={() => setFilter(c.key)}
   503→            aria-pressed={filter === c.key}
   504→            className={`bg-white rounded-2xl border p-4 flex items-center gap-3 text-left transition-colors cursor-pointer min-h-[44px] ${
   505→              filter === c.key ? 'border-sage bg-sage-light/10' : 'border-sage-light/20 hover:border-sage/40'
   506→            }`}
   507→          >
   508→            <span className={`w-11 h-11 rounded-xl flex items-center justify-center flex-shrink-0 ${c.tone}`}>{c.icon}</span>
   509→            <span className="min-w-0">
   510→              <span className="block text-xl font-bold text-brown-dark leading-none tabular-nums">{c.value}</span>
   511→              <span className="block text-[11px] font-semibold text-brown-light mt-1 truncate">{c.label}</span>
   512→            </span>
   513→          </button>
   514→        ))}
   515→      </div>
   516→
   517→      {/* ---------- زیرعنوان + جست‌وجو + فیلترها ---------- */}
   518→      <div className="flex flex-wrap items-center justify-between gap-3 mb-4">
   519→        <p className="text-xs text-brown-light bg-butter/20 rounded-xl px-4 py-2.5 flex-1 min-w-[240px]">{pa.subtitle}</p>
   520→        <input
   521→          type="search"
   522→          value={query}
   523→          onChange={(e) => setQuery(e.target.value)}
   524→          placeholder={pa.searchPlaceholder}
   525→          aria-label={pa.searchPlaceholder}
   526→          className={`${inputCls} max-w-xs`}
   527→        />
   528→      </div>

   935→                            {actionBusy && <Loader2 className="w-3.5 h-3.5 animate-spin" aria-hidden="true" />}
   936→                            {pa.approveYes}
   937→                          </button>
   938→                          <button
   939→                            type="button"
   940→                            onClick={() => setApproveStage(false)}
   941→                            disabled={actionBusy}
   942→                            className="bg-white border border-sage-light/40 text-brown px-4 py-2.5 rounded-xl text-xs font-semibold hover:border-sage transition-colors cursor-pointer disabled:opacity-60 min-h-[44px]"
   943→                          >
   944→                            {pa.close}
   945→                          </button>
   946→                        </div>
   947→                      </div>
   948→                    )}
   949→
   950→                    {/* گام ردّ + دلیل */}
   951→                    {rejectStage && (
   952→                      <div className="bg-peach-light/50 border border-peach/50 rounded-xl px-4 py-3.5" role="alertdialog" aria-label={pa.rejectTitle}>
   953→                        <p className="text-sm font-bold text-brown-dark mb-1">{pa.rejectTitle}</p>
   954→                        <p className="text-xs font-semibold text-brown mb-2">{pa.rejectReasonLabel}</p>
   955→                        <div className="flex flex-wrap gap-2 mb-3">
   956→                          {pa.rejectPresets.map((preset) => (
   957→                            <button
   958→                              key={preset}
   959→                              type="button"
   960→                              onClick={() => setRejectReason(preset)}
   961→                              className={`text-[11px] font-semibold px-3 py-1.5 rounded-full transition-colors cursor-pointer min-h-[32px] ${
   962→                                rejectReason === preset
   963→                                  ? 'bg-brown-dark text-white'
   964→                                  : 'bg-white border border-sage-light/40 text-brown hover:border-peach'
   965→                              }`}
   966→                            >
   967→                              {preset}
   968→                            </button>
   969→                          ))}
   970→                        </div>
   971→                        <textarea
   972→                          rows={3}
   973→                          value={rejectReason}
   974→                          onChange={(e) => setRejectReason(e.target.value)}
   975→                          placeholder={pa.rejectCustom}
   976→                          maxLength={500}
   977→                          aria-label={pa.rejectReasonLabel}
   978→                          className={`${inputCls} resize-none`}
   979→                        />
   980→                        <div className="flex flex-wrap items-center gap-2.5 mt-3">
   981→                          <button
   982→                            type="button"
   983→                            onClick={doReject}
   984→                            disabled={actionBusy || rejectReason.trim().length < 3}
   985→                            className="bg-red-500 text-white px-4 py-2.5 rounded-xl text-xs font-bold hover:bg-red-600 transition-colors cursor-pointer disabled:opacity-50 disabled:cursor-not-allowed inline-flex items-center gap-1.5 min-h-[44px]"
   986→                          >
   987→                            {actionBusy && <Loader2 className="w-3.5 h-3.5 animate-spin" aria-hidden="true" />}
   988→                            {pa.rejectSubmit}
   989→                          </button>
   990→                          <button
   991→                            type="button"
   992→                            onClick={() => {
   993→                              setRejectStage(false)
   994→                              setRejectReason('')
   995→                            }}
   996→                            disabled={actionBusy}
   997→                            className="bg-white border border-sage-light/40 text-brown px-4 py-2.5 rounded-xl text-xs font-semibold hover:border-sage transition-colors cursor-pointer disabled:opacity-60 min-h-[44px]"
   998→                          >
   999→                            {pa.close}
  1000→                          </button>
  1001→                        </div>
  1002→                      </div>
  1003→                    )}
  1004→                  </div>
  1005→                )}
  1006→
  1007→                {/* لغو سفارش — فقط PENDING */}
  1008→                {detail.order.status === 'PENDING' && (
  1009→                  <div className="mt-3">
  1010→                    <button
  1011→                      type="button"
  1012→                      onClick={doCancel}
  1013→                      disabled={actionBusy}
  1014→                      className="text-xs font-semibold text-brown-light hover:text-red-500 border border-sage-light/40 hover:border-red-200 rounded-xl px-4 py-2.5 transition-colors cursor-pointer disabled:opacity-60 min-h-[44px]"
  1015→                    >
  1016→                      {pa.cancelAction}
  1017→                    </button>
  1018→                  </div>
  1019→                )}
  1020→              </section>
  1021→
  1022→              {/* 🧵 زنجیرهٔ ممیزی */}
  1023→              <section>
  1024→                <h4 className="text-xs font-bold uppercase tracking-wider text-brown-light mb-2 flex items-center gap-1.5">
  1025→                  <Clock className="w-3.5 h-3.5" aria-hidden="true" /> {pa.blockEvents}
  1026→                </h4>
  1027→                {detail.events.length === 0 ? (
  1028→                  <p className="text-xs text-brown-light bg-cream/50 border border-dashed border-sage-light/40 rounded-xl px-4 py-5 text-center">
  1029→                    {pa.eventsEmpty}
  1030→                  </p>
  1031→                ) : (
  1032→                  <ul className="bg-white border border-sage-light/20 rounded-xl overflow-hidden max-h-64 overflow-y-auto ct-scroll-area">
  1033→                    {detail.events.map((e, i) => (
  1034→                      <li
  1035→                        key={e.id}
  1036→                        className={`px-4 py-2.5 flex flex-wrap items-center gap-x-3 gap-y-1 text-xs ${
  1037→                          i !== detail.events.length - 1 ? 'border-b border-sage-light/15' : ''
  1038→                        }`}
  1039→                      >
  1040→                        <span
  1041→                          className={`font-bold px-2 py-0.5 rounded-full text-[10px] whitespace-nowrap ${
  1042→                            e.kind === 'PAID' || e.kind === 'APPROVED'
  1043→                              ? 'bg-green-100 text-green-800'
  1044→                              : e.kind === 'REJECTED'
  1045→                                ? 'bg-peach-light/60 text-brown'
  1046→                                : 'bg-cream text-brown-light'
  1047→                          }`}
  1048→                        >
  1049→                          {e.kind}
  1050→                        </span>
  1051→                        <span className="flex-1 min-w-32 truncate text-brown-light" title={e.detail}>
  1052→                          {eventDetailLine(e.detail)}
  1053→                        </span>
  1054→                        <span className="text-brown-light/70 whitespace-nowrap">{fmtDateTime(e.createdAt)}</span>
  1055→                      </li>
  1056→                    ))}
  1057→                  </ul>
  1058→                )}
  1059→              </section>
  1060→            </div>
  1061→          )}
  1062→        </DialogContent>
  1063→      </Dialog>
  1064→
  1065→      {/* 🔍 لایت‌باکس تمام‌صفحهٔ رسید */}
  1066→      {lightbox && receiptObjUrl && isImageReceipt && (
  1067→        <div
  1068→          role="dialog"
  1069→          aria-modal="true"
  1070→          aria-label={pa.receiptView}
  1071→          className="fixed inset-0 bg-black/80 z-[70] flex items-center justify-center p-6 cursor-zoom-out"
  1072→          onClick={() => setLightbox(false)}
  1073→        >
  1074→          <img src={receiptObjUrl} alt={detail?.order.receiptFileName || 'receipt'} className="max-h-[90vh] max-w-[90vw] rounded-xl shadow-2xl" />
  1075→          <button
  1076→            type="button"
  1077→            onClick={() => setLightbox(false)}
  1078→            aria-label={pa.close}
  1079→            className="absolute top-5 right-5 w-10 h-10 rounded-full bg-white/90 text-brown-dark flex items-center justify-center hover:bg-white transition-colors cursor-pointer text-lg font-bold"
  1080→          >
  1081→            ✕
  1082→          </button>
  1083→          <p className="absolute bottom-5 left-1/2 -translate-x-1/2 text-white/70 text-xs">{pa.zoomHint}</p>
  1084→        </div>
  1085→      )}
  1086→    </div>
  1087→  )
  1088→}
