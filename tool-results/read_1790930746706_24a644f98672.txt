   300→// 🧾 سفارش‌های خرید کلاس
   301→// از فاز ۵۹ روش پرداختِ فعال «کارت بانکیِ دستی + رسید + تأیید ادمین» است
   302→// (paymentMethod = MANUAL_BANK_CARD)؛ ردیف‌های قدیمی با paymentMethod =
   303→// USDT_TRON (پرداخت آنلاین سابق) دست‌نخورده و حفظ شده‌اند.
   304→// مبلغ‌ها به «میکرو» ذخیره می‌شوند (۱ واحد = ۱٬۰۰۰٬۰۰۰) تا هیچ خطای اعشاری رخ ندهد.
   305→// وضعیت‌ها:
   306→//   جریانِ دستی: PENDING (پرداخت‌نشده) → RECEIPT_SUBMITTED (رسید در انتظار بررسی)
   307→//                → PAID (تأیید ادمین) | REJECTED (ردّ ادمین) → [ارسال دوبارهٔ رسید]
   308→//   CANCELLED (لغو ادمین) — جریانِ تاریخیِ USDT: DETECTED | UNDERPAID | EXPIRED
   309→// txHash یکتاست → هر تراکنش قدیمی روی blockchain فقط یک‌بار مصرف شده (ضد replay)
   310→// receiptId یکتا و غیرقابل‌حدس است — فایل رسید هرگز در پوشهٔ عمومی نیست و فقط
   311→// از مسیر احرازِ مالک/ادمین سرو می‌شود.
   312→// ---------------------------------------------------------------------------
   313→model UsdtOrder {
   314→  id             String    @id @default(cuid())
   315→  ref            String    @unique // شناسهٔ عمومی سفارش (CT-XXXXXXXX) — در URL صفحهٔ پرداخت
   316→  userId         String? // کاربرِ صاحب سفارش (سیستم حساب‌ها) — سفارش‌های مهمانِ قدیمی null می‌ماند
   317→  user           User?     @relation(fields: [userId], references: [id], onDelete: SetNull)

   504→// هر رکورد = یک جلسهٔ واقعی برای یک دانش‌پذیر: شروع و پایان به‌صورت «لحظهٔ

   594→  createdAt  DateTime  @default(now())
   595→
   596→  @@index([userId, readAt])
   597→  @@index([userId, category, createdAt])
   598→}

   601→// 💾 لاگ پشتیبان‌گیری دیتابیس (فاز ۵۲) — هر عملیات Backup یک ردیف
   602→// status: STARTED (شروع شد) → SUCCESS | FAILED
   603→// trigger: SCHEDULED (زمان‌بند خودکار) | MANUAL (دکمهٔ ادمین) | PRE_RESTORE (پشتیبان ایمنی قبل از Restore)
   604→// هیچ رازی (کلید/رمز/seed) در این جدول یا لاگ‌ها نوشته نمی‌شود.
   605→// ---------------------------------------------------------------------------
   606→model BackupLog {
   607→  id          String    @id @default(cuid())
   608→  filename    String // نام فایل پشتیبان (backup-<timestamp>-<trigger>.db)
   609→  filePath    String // مسیر مطلق روی سرور — هرگز URL عمومی نیست
   610→  sizeBytes   Int? // حجم فایل پس از اتمام موفق
   611→  status      String    @default("STARTED") // STARTED | SUCCESS | FAILED
   612→  trigger     String    @default("MANUAL") // SCHEDULED | MANUAL | PRE_RESTORE
   613→  startedAt   DateTime  @default(now())
   614→  completedAt DateTime?
   615→  durationMs  Int? // مدت اجرا — برای پایش سلامت
   616→  error       String? // پیام کوتاه خطا — بدون محتوای حساس دیتابیس
   617→
   618→  @@index([status, startedAt])
   619→  @@index([trigger, startedAt])
   620→}
   621→
   622→// ---------------------------------------------------------------------------
   623→// 🚨 لاگ خطای سمت سرور (فاز ۵۳ — بند ۱۲) — پایش خطاهای مهم اپلیکیشن
   624→// فقط «پیام امن» ذخیره می‌شود: هرگز رمز/توکن/کلید/stack کامل/محتوای حساس.
   625→// category: API | DATABASE | PAYMENT | ORDER | AUTH | BACKUP | RESTORE | SERVER | VALIDATION
   626→// ---------------------------------------------------------------------------
   627→model ErrorLog {
   628→  id         String   @id @default(cuid())
   629→  category   String // API | DATABASE | PAYMENT | ORDER | AUTH | BACKUP | RESTORE | SERVER | VALIDATION
   630→  message    String // پیام کوتاه و امن — بدون راز و بدون stack کامل
   631→  context    String   @default("") // JSON: path، method، routeKind — بدون PII اضافه
   632→  statusCode Int? // کد وضعیت HTTP در صورت وجود
   633→  refId      String? // شناسهٔ رکورد مرتبط (orderRef، backupId و…)
   634→  createdAt  DateTime @default(now())
   635→
   636→  @@index([category, createdAt])
   637→}

   639→// ---------------------------------------------------------------------------
   640→// 🧾 گزارش فعالیت ادمین (فاز ۵۳ — بند ۱۴) — append-oriented؛ رکوردها هرگز
   641→// ویرایش/حذف نمی‌شوند. هرگز راز/کلید/رمز در meta ذخیره نمی‌شود.
   642→// ---------------------------------------------------------------------------
   643→model AuditLog {
   644→  id         String   @id @default(cuid())
   645→  actor      String // نام کاربر ادمین (مثل admin) — بدون توکن
   646→  action     String // discount.create | discount.update | discount.archive | discount.restore | class.create | class.update | class.status | class.delete | order.cancel | registration.status | schedule.propose | wallet.save | wallet.remove | credentials.update | settings.update | backup.create | backup.restore | backup.settings
   647→  targetType String // discount | class | order | registration | schedule | wallet | settings | backup
   648→  targetId   String   @default("") // شناسه/کد/ref رکورد هدف
   649→  meta       String   @default("") // JSON متادیتای امن (قبل/بعد — بدون راز)
   650→  createdAt  DateTime @default(now())
   651→
   652→  @@index([createdAt])
   653→  @@index([action, createdAt])
   654→}
