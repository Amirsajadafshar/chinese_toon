   235→   235→export async function createOrder(input: CreateOrderInput): Promise<PublicOrder> {
   236→   236→  // 💳 پرداخت دستی باید از پنل ادمین فعال و کارت تنظیم شده باشد — fail-closed
   237→   237→  const bank = await getBankCardSettings()
   238→   238→  if (!isBankCardReady(bank)) {
   239→   239→    console.error('[payments] create blocked — manual bank-card payment is not configured/enabled')
   240→   240→    throw new PaymentNotConfiguredError('Manual bank-card payment is not configured on this server')
   241→   241→  }
   242→   242→
   243→   243→  // 🎓 مبلغ مرجع از جدول CourseClass — با موتور قیمت‌گذاری:
   244→   244→  //   base = pricePerSession × sessions، تخفیف خودکار بسته (پله‌ها) و کد تخفیف
   245→   245→  //   همه سمت سرور. مرورگر فقط productId و «کدِ» تخفیف را می‌فرستد.
   246→   246→  const product = await findProductAsync(input.productId)
   247→   247→  if (!product) throw new ProductNotFoundError(`Unknown product: ${input.productId}`)
   248→   248→
   249→   249→  // ردیف کامل کلاس از DB — برای محاسبهٔ مقتدر (fallback فایل = بدون کد)
   250→   250→  const { findBookableClassRowByProductId } = await import('@/lib/classes/store')
   251→   251→  const clsRow = await findBookableClassRowByProductId(input.productId)
   252→   252→  let finalUsd = product.amountUsd
   253→   253→  let breakdown: Awaited<ReturnType<typeof calculatePrice>>['breakdown'] | null = null
   254→   254→  if (clsRow) {
   255→   255→    const calc = await calculatePrice(clsRow, { code: input.discountCode, email: input.contactEmail })
   256→   256→    breakdown = calc.breakdown
   257→   257→    if (breakdown.final > 0) finalUsd = breakdown.final
   258→   258→    if (calc.rejection && input.discountCode) {
   259→   259→      console.info(`[payments] discount code "${input.discountCode}" rejected (${calc.rejection}) — order continues without code discount`)
   260→   260→    }
   261→   261→  }
   262→   262→
   263→   263→  // 💳 مبلغِ واریزِ مشتری = دقیقاً مبلغ نهایی — بدون هیچ افست/اسلات
   264→   264→  const baseMicro = usdToMicro(finalUsd)
   265→   265→  const contactName = (input.contactName || '').slice(0, 80)
   266→   266→  const contactEmail = (input.contactEmail || '').slice(0, 120)
   267→   267→
   268→   268→  // حلقهٔ ضد-مسابقه: اگر درخواست موازی‌ای در حال نوشتن باشد، SQLite با P2034
   269→   269→  // رد می‌کند → با دید تازه دوباره تلاش می‌کنیم (idempotency داخل تراکنش است).
   270→   270→  let order: Awaited<ReturnType<typeof buildOrderTx>> | null = null
   271→   271→  for (let attempt = 0; attempt < 5 && !order; attempt++) {
   272→   272→    try {
   273→   273→      order = await buildOrderTx(input, product, baseMicro, contactName, contactEmail, breakdown)
   274→   274→    } catch (e) {
   275→   275→      // 🎟️ کد بین پیش‌نمایش و این لحظه تمام/منقضی/غیرفعال شد:
   276→   276→      // سفارش «بی‌صدا» با قیمت کامل ساخته نمی‌شود — ۴۰۹ شفاف به مشتری.
   277→   277→      if (e instanceof DiscountExhaustedError) throw e
   278→   278→      // 🛡️ ضد ارسال تکراری — سفارشِ بازِ همان محصول در ۲ دقیقهٔ اخیر:
   279→   279→      // همان سفارش موجود برمی‌گردد (پاسخ idempotent؛ بدون ساخت رکورد تازه)
   280→   280→      if (e instanceof DuplicateRecentOrderError) {
   281→   281→        const existing = await getOrderByRef(e.existingRef)
   282→   282→        if (existing) {
   283→   283→          await logEvent('DUPLICATE_SUPPRESSED', existing.ref, undefined, {
   284→   284→            windowMs: ORDER_IDEMPOTENCY_MS,
   285→   285→            note: 'Duplicate order creation suppressed — existing open order returned',
   286→   286→          })
   287→   287→          return publicOrder(existing, product.label)
   288→   288→        }
   289→   289→        // سفارشِ موجود در همین فاصله حذف/بسته شده → ادامه به ساختِ عادی
   290→   290→        continue
   291→   291→      }
   292→   292→      if (isWriteConflict(e)) continue // تلاش مجدد با دید تازه
   293→   293→      throw e
   294→   294→    }
   295→   295→  }
   296→   296→  if (!order) throw new OrderBusyError('Could not allocate an order — please try again shortly')
   297→   297→
   298→   298→  await logEvent('CREATED', order.ref, undefined, {
   299→   299→    product: product.id,
   300→   300→    amountUsd: microToDisplay(order.expectedMicro),
   301→   301→    paymentMethod: 'MANUAL_BANK_CARD',
   302→   302→    ...(breakdown ? { base: breakdown.base.toFixed(2), tierPercent: String(breakdown.tierPercent), code: breakdown.code ?? '-', discount: breakdown.codeDiscount.toFixed(2) } : {}),
   303→   303→  })
   304→   304→
   305→   305→  // 🔔 اعلان ORDER_CREATED — فقط از رویداد واقعی backend
   306→   306→  if (input.userId) {
   307→   307→    notifyUser({
   308→   308→      userId: input.userId,
   309→   309→      category: 'order',
   310→   310→      kind: 'ORDER_CREATED',
   311→   311→      title: 'Order created',
   312→   312→      body: `Your order ${order.ref} for “${product.label}” (${microToDisplay(order.expectedMicro)} USD) has been created. Please transfer the exact amount to the bank card shown on the payment page, then upload your payment receipt so we can verify it.`,
   313→   313→      orderRef: order.ref,
   314→   314→    })
   315→   315→  }
   316→   316→
   317→   317→  return publicOrder(order, product.label)
   318→   318→}

   320→   320→/** یک تلاش کامل ساخت سفارش داخل یک تراکنش دیتابیس — جریانِ پرداخت دستی */
   321→   321→async function buildOrderTx(
   322→   322→  input: CreateOrderInput,
   323→   323→  product: PaymentProduct,
   324→   324→  baseMicro: number,
   325→   325→  contactName: string,
   326→   326→  contactEmail: string,
   327→   327→  breakdown: Awaited<ReturnType<typeof calculatePrice>>['breakdown'] | null
   328→   328→) {
   329→   329→  const { userId } = input
   330→   330→  return database.$transaction(async (tx) => {
   331→   331→    // 🛡️ ضد ارسال تکراری (idempotency) — داخل همان تراکنش: اگر همین کاربر همان
   332→   332→    // محصول را لحظاتی پیش سفارش داده باشد و هنوز باز باشد، به‌جای رکورد تازه همان
   333→   333→    // سفارش برمی‌گرداند (دابل‌کلیک هرگز دو سفارش نمی‌سازد).
   334→   334→    if (userId) {
   335→   335→      const dup = await tx.usdtOrder.findFirst({
   336→   336→        where: {
   337→   337→          userId,
   338→   338→          productId: product.id,
   339→   339→          status: { in: [...ACTIVE_STATUSES] },
   340→   340→          createdAt: { gte: new Date(Date.now() - ORDER_IDEMPOTENCY_MS) },
   341→   341→        },
   342→   342→        orderBy: { createdAt: 'desc' },
   343→   343→        select: { ref: true },
   344→   344→      })
   345→   345→      if (dup) throw new DuplicateRecentOrderError(dup.ref)
   346→   346→    }
   347→   347→
   348→   348→    // سقف سفارش باز برای هر IP
   349→   349→    const openCount = await tx.usdtOrder.count({
   350→   350→      where: { clientIp: input.clientIp, status: { in: [...ACTIVE_STATUSES] } },
   351→   351→    })
   352→   352→    if (openCount >= MAX_OPEN_ORDERS_PER_IP) throw new OrderLimitError('Too many open orders')
   353→   353→
   354→   354→    // 🛡️ سقف مستقل برای هر کاربر (ضد چرخش IP برای دورزدن سقف IP)
   355→   355→    if (userId) {
   356→   356→      const openUserCount = await tx.usdtOrder.count({
   357→   357→        where: { userId, status: { in: [...ACTIVE_STATUSES] } },
   358→   358→      })
   359→   359→      if (openUserCount >= MAX_OPEN_ORDERS_PER_USER) throw new OrderLimitError('Too many open orders')
   360→   360→    }
   361→   361→
   362→   362→    const ref = generateOrderRef()
   363→   363→    const expiresAt = new Date(Date.now() + MANUAL_ORDER_TTL_MS)
   364→   364→    const created = await tx.usdtOrder.create({
   365→   365→      data: {
   366→   366→        ref,
   367→   367→        userId: userId ?? null,
   368→   368→        productId: product.id,
   369→   369→        productTitle: product.label,
   370→   370→        contactName,
   371→   371→        contactEmail,
   372→   372→        expectedMicro: baseMicro, // 💳 واریز دقیقِ مبلغ نهایی — بدون افست
   373→   373→        currency: 'USD', // 💳 پرداخت بانکی — نمایش دلار
   374→   374→        network: 'BANK', // نشانهٔ «خارج از بلاکچین» — شبکهٔ قدیمی سفارش‌های USDT حفظ شده
   375→   375→        paymentMode: 'manual',
   376→   376→        paymentAddress: '', // 🧾 در پرداخت بانکی آدرسِ زنجیره‌ای وجود ندارد
   377→   377→        addressIndex: null,
   378→   378→        clientIp: input.clientIp,
   379→   379→        paymentMethod: 'MANUAL_BANK_CARD',
   380→   380→        status: 'PENDING',
   381→   381→        expiresAt,
   382→   382→        // 💰 اسنپ‌شات قیمت‌گذاری — سفارش تاریخی هرگز از پیکربندی آینده پیروی نمی‌کند
   383→   383→        // 🎟️ نوع/مقدار کد هم اسنپ‌شات می‌شود (دقت تاریخی بعد از ویرایش ادمین)
   384→   384→        ...(breakdown
   385→   385→          ? {
   386→   386→              pricePerSession: breakdown.pricePerSession,
   387→   387→              packageSessions: breakdown.sessions,
   388→   388→              baseAmount: breakdown.base,
   389→   389→              tierPercent: breakdown.tierPercent,
   390→   390→              discountCode: breakdown.code,
   391→   391→              discountAmount: breakdown.codeDiscount,
   392→   392→              discountType: breakdown.codeType,
   393→   393→              discountValue: breakdown.codeValue,
   394→   394→            }
   395→   395→          : {}),
   396→   396→      },
   397→   397→    })
   398→   398→
   399→   399→    // 🎟️ ثبت استفادهٔ کد داخل «همان تراکنش» سفارش:
   400→   400→    //  • شمارش maxUses/perCustomer + درج Redemption اتمی است — سفارش بدون ثبت
   401→   401→    //    استفاده هرگز commit نمی‌شود و برعکس.
   402→   402→    //  • قواعد زمان‌محور/وضعیت‌محور (active/startsAt/endsAt) دوباره با «زمان سرور»
   403→   403→    //    چک می‌شوند — مرورگر و پیش‌نمایش قبلی هیچ اعتباری ندارند.
   404→   404→    if (breakdown?.discountId && breakdown.codeDiscount > 0 && breakdown.code) {
   405→   405→      const dRow = await tx.discount.findUnique({ where: { id: breakdown.discountId } })
   406→   406→      const now = new Date() // ⏰ زمان سرور — مرورگر هرگز اعتبار کد را تعیین نمی‌کند
   407→   407→      if (!dRow || !dRow.active) throw new DiscountExhaustedError(breakdown.code)
   408→   408→      if (dRow.startsAt && dRow.startsAt > now) throw new DiscountExhaustedError(breakdown.code)
   409→   409→      if (dRow.endsAt && dRow.endsAt < now) throw new DiscountExhaustedError(breakdown.code)
   410→   410→      const usedTotal = await tx.discountRedemption.count({ where: { discountId: dRow.id } })
   411→   411→      if (dRow.maxUses !== null && usedTotal >= dRow.maxUses) throw new DiscountExhaustedError(breakdown.code)
   412→   412→      const redemptionEmail = contactEmail.toLowerCase()
   413→   413→      if (redemptionEmail) {
   414→   414→        const usedBy = await tx.discountRedemption.count({
   415→   415→          where: { discountId: dRow.id, userEmail: redemptionEmail },
   416→   416→        })
   417→   417→        if (usedBy >= Math.max(1, dRow.perCustomer)) throw new DiscountExhaustedError(breakdown.code)
   418→   418→      }
   419→   419→      await tx.discountRedemption.create({
   420→   420→        data: {
   421→   421→          discountId: dRow.id,
   422→   422→          orderRef: ref,
   423→   423→          userEmail: redemptionEmail,
   424→   424→          amount: breakdown.codeDiscount,
   425→   425→        },
   426→   426→      })
   427→   427→    }
   428→   428→
   429→   429→    return created
   430→   430→  })
   431→   431→}

   442→   442→export async function buildPublicStatus(ref: string): Promise<PublicOrder | null> {

   445→   445→  // 📊 وضعیت نمایشی مشتقِ سمت سرور است — PENDINGِ منقضی = EXPIRED حتی اگر
   446→   446→  // سوئیپر هنوز رکورد را به‌روز نکرده باشد؛ مشتری هرگز صفحهٔ قابل‌پرداختِ
   447→   447→  // سفارشِ منقضی نمی‌بیند. RECEIPT_SUBMITTED هرگز منقضی نمایش داده نمی‌شود.
   448→   448→  const derived = deriveOrderStatus(o)
   449→   449→  // محصولِ کلاس ممکن است بعداً حذف/غیرفعال شده باشد — اسنپ‌شات سفارش (productTitle)

   451→   451→  const product = await findProductAsync(o.productId)
   452→   452→  return publicOrder({ ...o, status: derived }, product?.label ?? o.productTitle)
   453→   453→}

   459→   459→export type SubmitReceiptResult =
   460→   460→  | { ok: true; resubmitted: boolean }
   461→   461→  | { ok: false; reason: 'NOT_FOUND' | 'NOT_OWNER' | 'INVALID_STATE' | 'ALREADY_SUBMITTED' | 'CONFLICT' }
   462→   462→
   463→   463→export async function submitReceipt(input: {
   464→   464→  ref: string
   465→   465→  viewerUserId: string
   466→   466→  receiptId: string
   467→   467→  mime: ReceiptMime
   468→   468→  size: number
   469→   469→  fileName: string
   470→   470→  fileData: Buffer
   471→   471→}): Promise<SubmitReceiptResult> {
   472→   472→  const order = await getOrderByRef(input.ref)
   473→   473→  if (!order) return { ok: false, reason: 'NOT_FOUND' }
   474→   474→  // 🔒 فقط مالکِ سشن — پاسخِ «پیدا نشد» برای سفارشِ دیگران (بدون نشت وجودش)
   475→   475→  if (!order.userId || order.userId !== input.viewerUserId) return { ok: false, reason: 'NOT_FOUND' }
   476→   476→
   477→   477→  const derived = deriveOrderStatus(order)
   478→   478→  const resubmitted = derived === 'REJECTED'
   479→   479→  if (derived === 'RECEIPT_SUBMITTED') {
   480→   480→    // 🛡️ ضد ارسال تکراری — رسیدِ در انتظار بررسی؛ دوباره پذیرفته نمی‌شود
   481→   481→    return { ok: false, reason: 'ALREADY_SUBMITTED' }
   482→   482→  }
   483→   483→  if (derived !== 'PENDING' && derived !== 'REJECTED') {
   484→   484→    return { ok: false, reason: 'INVALID_STATE' }
   485→   485→  }
   486→   486→
   487→   487→  // فایل اول ذخیره می‌شود (شناسهٔ یکتا — بدون تداخل)؛ اگر گارد وضعیت شکست بخورد
   488→   488→  // فایل یتیم حذف می‌شود. گارد، هم‌زمانیِ دو ارسال موازی را می‌گیرد.
   489→   489→  await saveReceiptFile(input.receiptId, input.mime, input.fileData)
   490→   490→  const res = await database.usdtOrder
   491→   491→    .updateMany({
   492→   492→      where: {
   493→   493→        ref: input.ref,
   494→   494→        status: { in: ['PENDING', 'REJECTED'] },
   495→   495→        OR: [{ status: 'REJECTED' }, { expiresAt: { gt: new Date() } }],
   496→   496→      },
   497→   497→      data: {
   498→   498→        status: 'RECEIPT_SUBMITTED',
   499→   499→        receiptStatus: 'PENDING_REVIEW',
   500→   500→        receiptId: input.receiptId,
   501→   501→        receiptMime: input.mime,
   502→   502→        receiptSize: input.size,
   503→   503→        receiptFileName: input.fileName,
   504→   504→        receiptSubmittedAt: new Date(),
   505→   505→        rejectionReason: null, // ارسال دوباره = دلیل قبلی پاک می‌شود
   506→   506→        reviewedAt: null,
   507→   507→        reviewedBy: null,
   508→   508→        ...(resubmitted ? { receiptResubmits: { increment: 1 } } : {}),
   509→   509→      },
   510→   510→    })
   511→   511→    .then((r) => r.count)
   512→   512→    .catch(() => 0)
   513→   513→
   514→   514→  if (res !== 1) {
   515→   515→    await deleteReceiptFile(input.receiptId)
   516→   516→    return { ok: false, reason: 'CONFLICT' }
   517→   517→  }
   518→   518→
   519→   519→  // حذف فایل رسید قبلی (ارسال دوباره پس از رد) — بعد از موفقیتِ به‌روزرسانی
   520→   520→  if (order.receiptId && order.receiptId !== input.receiptId) {
   521→   521→    await deleteReceiptFile(order.receiptId)
   522→   522→  }
   523→   523→
   524→   524→  await logEvent(resubmitted ? 'RECEIPT_RESUBMITTED' : 'RECEIPT_SUBMITTED', input.ref, undefined, {
   525→   525→    size: input.size,
   526→   526→    mime: input.mime,
   527→   527→    resubmits: (order.receiptResubmits ?? 0) + (resubmitted ? 1 : 0),
   528→   528→  })
   529→   529→
   530→   530→  // 🔔 اعلان مشتری: رسید دریافت شد و در انتظار بررسی است (وضعیت هرگز «پرداخت‌شده» نیست)
   531→   531→  notifyOrderEvent(input.ref, {
   532→   532→    kind: 'RECEIPT_RECEIVED',
   533→   533→    category: 'payment',
   534→   534→    title: 'Receipt submitted',
   535→   535→    body: `Your payment receipt for order ${input.ref} has been submitted and is now waiting for verification. We'll notify you as soon as our team reviews it.`,
   536→   536→  })
   537→   537→
   538→   538→  return { ok: true, resubmitted }
   539→   539→}

   607→   607→}

   622→   622→export async function expireStaleOrders(network: string = 'BANK'): Promise<number> {
   623→   623→  const stale = await database.usdtOrder.findMany({
   624→   624→    where: { status: 'PENDING', expiresAt: { lt: new Date() }, network: networkFilter(network) },
   625→   625→    select: { ref: true, userId: true },
   626→   626→  })
   627→   627→  let n = 0
   628→   628→  for (const o of stale) {
   629→   629→    // گارد وضعیت‌محور: فقط PENDING منقضی می‌شود؛ PAID/RECEIPT_SUBMITTED هم‌زمان دست‌نخورده می‌مانند
   630→   630→    const res = await database.usdtOrder
   631→   631→      .updateMany({ where: { ref: o.ref, status: 'PENDING' }, data: { status: 'EXPIRED' } })
   632→   632→      .then((r) => r.count)
   633→   633→      .catch(() => 0)
   634→   634→    if (res === 1) {
   635→   635→      n++
   636→   636→      await logEvent('EXPIRED', o.ref)
   637→   637→      // 🎟️ آزادسازی سهمیهٔ کد تخفیف: سفارشِ منقضی هرگز پرداخت نشده،
   638→   638→      // پس استفاده‌اش نباید در maxUses/perCustomer حساب شود.
   639→   639→      await database.discountRedemption.deleteMany({ where: { orderRef: o.ref } }).catch(() => 0)
   640→   640→      // 🔔 اعلان ORDER_EXPIRED — انقضای سمت سرور است، نه تصمیم مرورگر
   641→   641→      if (o.userId) {
   642→   642→        notifyUser({
   643→   643→          userId: o.userId,
   644→   644→          category: 'order',
   645→   645→          kind: 'ORDER_EXPIRED',
   646→   646→          title: 'Order expired',
   647→   647→          body: `Your order ${o.ref} expired because the payment deadline passed. You can safely create a new order anytime — no payment was deducted.`,
   648→   648→          orderRef: o.ref,
   649→   649→        })
   650→   650→      }
   651→   651→    }
   652→   652→  }
   653→   653→  return n
   654→   654→}
