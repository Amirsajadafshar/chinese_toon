     3→     3→//
     4→     4→// ریشهٔ واقعی «Could not create the account (Ref: …)» و «لاگین نمی‌شود» روی
     5→     5→// محیط استقرار: فایل SQLite تولید یا «خالی» است (بدون هیچ جدولی — Prisma خودش
     6→     6→// هرگز جدول نمی‌سازد؛ db push یک گام دستی CLI است) یا از یک بکاپِ قدیمی با
     7→     7→// اسکیمای کهنه بازیابی شده است (ستون‌های جدید مثل dateOfBirth/uniqueCode غایب‌اند).
     8→     8→// نتیجه: هر کوئری Prisma با P2021 (جدول غایب) یا P2022 (ستون غایب) خطا می‌دهد و

    12→    12→//   • جدولی را drop نمی‌کند
    13→    13→//   • ستونی را حذف یا نوعش را تغییر نمی‌دهد
    14→    14→//   • هیچ ردیف داده‌ای را دست نمی‌زند
    15→    15→// فقط: جدول‌های غایب را با شکل کامل فعلی می‌سازد، ستون‌های غایب جدول‌های موجود
    16→    16→// را با DEFAULT امن اضافه می‌کند و ایندکس‌های یکتا/جست‌وجو را تضمین می‌کند.
    17→    17→// اجرای دوباره کاملاً بی‌اثر (idempotent) است.

    24→    24→/** تشخیص خطای ناهم‌خوانی اسکیما (جدول/ستون غایب) از روی کد Prisma */
    25→    25→export function isSchemaDriftError(e: unknown): boolean {
    26→    26→  const code = (e as { code?: string } | null)?.code
    27→    27→  return code === 'P2021' || code === 'P2022'
    28→    28→}

   643→   643→export interface EnsureSchemaResult {
   644→   644→  createdTables: string[]
   645→   645→  addedColumns: string[]
   646→   646→  ensuredIndexes: number
   647→   647→  errors: string[]
   648→   648→}
   649→   649→
   650→   650→async function runEnsureSchema(): Promise<EnsureSchemaResult> {
   651→   651→  const result: EnsureSchemaResult = { createdTables: [], addedColumns: [], ensuredIndexes: 0, errors: [] }
   652→   652→
   653→   653→  // فهرست جدول‌های پیش از اجرا — برای گزارش دقیق «چه جدولی ساخته شد»
   654→   654→  let before: Set<string>
   655→   655→  try {
   656→   656→    const rows = await db.$queryRawUnsafe<Array<{ name: string }>>(`SELECT name FROM sqlite_master WHERE type='table'`)
   657→   657→    before = new Set(Array.isArray(rows) ? rows.map((r) => String(r.name)) : [])
   658→   658→  } catch {
   659→   659→    before = new Set()
   660→   660→  }
   661→   661→
   662→   662→  // ۱) جدول‌های غایب ساخته می‌شوند (IF NOT EXISTS — اجرای دوباره بی‌اثر)
   663→   663→  for (const ddl of CREATE_TABLES) {
   664→   664→    const name = ddl.match(/CREATE TABLE IF NOT EXISTS "([^"]+)"/)?.[1] ?? ''
   665→   665→    try {
   666→   666→      await db.$executeRawUnsafe(ddl)
   667→   667→      if (name && !before.has(name)) result.createdTables.push(name)
   668→   668→    } catch (e) {
   669→   669→      result.errors.push(`create-table: ${e instanceof Error ? e.message : String(e)}`.slice(0, 200))
   670→   670→    }
   671→   671→  }
   672→   672→
   673→   673→  // ۲) ایندکس‌ها تضمین می‌شوند — شکست یکی (مثلاً دادهٔ تکراری قدیمی) بقیه را نمی‌شکند
   674→   674→  for (const ddl of CREATE_INDEXES) {
   675→   675→    try {
   676→   676→      await db.$executeRawUnsafe(ddl)
   677→   677→      result.ensuredIndexes++
   678→   678→    } catch (e) {
   679→   679→      result.errors.push(`index: ${e instanceof Error ? e.message : String(e)}`.slice(0, 200))
   680→   680→    }
   681→   681→  }
   682→   682→
   683→   683→  // ۳) ستون‌های غایب جدول‌های موجود (بازیابی‌شده از بکاپ کهنه) اضافه می‌شوند
   684→   684→  for (const [table, statements] of Object.entries(ALTER_COLUMNS)) {
   685→   685→    let existing: string[] = []
   686→   686→    try {
   687→   687→      const rows = await db.$queryRawUnsafe<Array<{ name: string }>>(`PRAGMA table_info("${table}")`)
   688→   688→      existing = Array.isArray(rows) ? rows.map((r) => String(r.name)) : []
   689→   689→    } catch {
   690→   690→      continue // جدول اصلاً موجود نیست — CREATE بالا هندل کرده
   691→   691→    }
   692→   692→    if (existing.length === 0) continue
   693→   693→    for (const stmt of statements) {
   694→   694→      const col = stmt.match(/ADD COLUMN\s+"([^"]+)"/)?.[1]
   695→   695→      if (!col || existing.includes(col)) continue
   696→   696→      try {
   697→   697→        await db.$executeRawUnsafe(stmt)
   698→   698→        result.addedColumns.push(`${table}.${col}`)
   699→   699→      } catch (e) {
   700→   700→        result.errors.push(`alter ${table}.${col}: ${e instanceof Error ? e.message : String(e)}`.slice(0, 200))
   701→   701→      }
   702→   702→    }
   703→   703→  }
   704→   704→
   705→   705→  return result
   706→   706→}

   708→   708→// ---------------------------------------------------------------------------
   709→   709→// درگاه عمومی — dedupe هم‌زمانی + دفترچهٔ یک‌بار در هر پروسه (مگر اجبار)
   710→   710→// ---------------------------------------------------------------------------
   711→   711→
   712→   712→let inFlight: Promise<EnsureSchemaResult> | null = null
   713→   713→let lastRunAt = 0
   714→   714→const RE_RUN_AFTER_MS = 60_000 // پس از یک اجرای موفق، حداکثر یک‌بار در دقیقه دوباره
   715→   715→
   716→   716→export async function ensureSchema(opts?: { force?: boolean; reason?: string }): Promise<EnsureSchemaResult> {
   717→   717→  const now = Date.now()
   718→   718→  if (!opts?.force && inFlight) return inFlight
   719→   719→  if (!opts?.force && now - lastRunAt < RE_RUN_AFTER_MS) {
   720→   720→    return { createdTables: [], addedColumns: [], ensuredIndexes: 0, errors: [] }
   721→   721→  }
   722→   722→  inFlight = runEnsureSchema()
   723→   723→    .then((r) => {
   724→   724→      lastRunAt = Date.now()
   725→   725→      if (r.createdTables.length > 0 || r.addedColumns.length > 0) {
   726→   726→        console.warn(
   727→   727→          `[ensure-schema] 🔧 self-healing applied (${opts?.reason ?? 'runtime'}): ` +
   728→   728→            `tables=[${r.createdTables.join(', ') || 'none'}] columns=[${r.addedColumns.join(', ') || 'none'}]` +
   729→   729→            (r.errors.length ? ` errors=${r.errors.length}` : '')
   730→   730→        )
   731→   731→      }
   732→   732→      return r
   733→   733→    })
   734→   734→    .finally(() => {
   735→   735→      inFlight = null
   736→   736→    })
   737→   737→  return inFlight
   738→   738→}
